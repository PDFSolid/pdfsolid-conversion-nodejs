PDFSolid Conversion Node.js API Documentation

This API reference documents the Node.js binding for PDFSolid Conversion SDK. The package is loaded with require("pdfsolidconversion") and exposes a thin JavaScript wrapper over a Node-API native addon.

The Node.js SDK supports Linux x86_64 and Windows x86_64. Public types are available from lib/index.d.ts.

Basic Usage

const sdk = require("pdfsolidconversion");

const licenseCode = sdk.licenseVerify("YOUR_LICENSE", "", "com.example.app");
if (licenseCode !== 0) {
  throw new Error(`license verification failed: ${licenseCode}`);
}

sdk.initialize();
try {
  sdk.setLogger(false, true);
  console.log(sdk.getVersion());

  const result = sdk.startPDFToWord(
    "/path/to/input.pdf",
    "",
    "/path/to/output.docx",
    {
      enableOcr: false,
      transparentText: true,
      pageRanges: "1-3",
      languages: [1]
    },
    {
      onProgress(currentPage, totalPage) {
        console.log(`progress: ${currentPage}/${totalPage}`);
      },
      isCancelled() {
        return false;
      }
    }
  );

  if (result !== 0) {
    throw new Error(`conversion failed: ${result}`);
  }
} finally {
  sdk.release();
}
